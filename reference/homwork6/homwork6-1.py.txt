#QUESTION2
X=df.iloc[:,:-1]
X=scale(X)
p=PCA()
p.fit(X)
W=p.components_.T
y=p.fit_transform(X)

W[:,0:3]
pd.DataFrame(W[:,:3],index=df.columns[:-1],columns=['W1','W2','W3'])
#Compute the explained variability by the PC scores
pd.DataFrame(p.explained_variance_ratio_,index=range(1,7),columns=['Explained Variability'])
pd.DataFrame(p.explained_variance_ratio_.cumsum(),index=range(1,7),columns=['Explained Variability'])
#Get the scree plot
plt.figure(2)
plt.bar(range(1,7),p.explained_variance_,color="blue",edgecolor="Red")
plt.figure(3)
#plt.bar(range(1,7),p.explained_variance_ratio_.cumsum(),color="blue",edgecolor="Red")
plt.errorbar(range(1,7),p.explained_variance_ratio_.cumsum())
